        function saveToStorage() {
            try {
                const data = {
                    categoriesData,
                    divisionsData,
                    clubMemberDatabase,
                    registrationOrders,
                    chairConfig,
                    chairPaymentRecords,
                    emailInboxLogs,
                    lastUpdated: new Date().toISOString()
                };
                localStorage.setItem(LOCAL_STORAGE_KEY, JSON.stringify(data));
            } catch (e) {
                console.warn("無法寫入 LocalStorage", e);
            }
        }

        function loadFromStorage() {
            try {
                const stored = localStorage.getItem(LOCAL_STORAGE_KEY);
                if (stored) {
                    const parsed = JSON.parse(stored);
                    if (parsed.categoriesData) {
                        categoriesData = parsed.categoriesData.map(c => {
                            if (c.id === 'ryla' || c.id === 'polio') {
                                return { ...c, allowAddSubEvents: false };
                            }
                            return { ...c, allowAddSubEvents: c.allowAddSubEvents !== false };
                        });
                    }
                    if (parsed.divisionsData && Array.isArray(parsed.divisionsData)) {
                        divisionsData = parsed.divisionsData;
                    }
                    if (parsed.clubMemberDatabase) clubMemberDatabase = parsed.clubMemberDatabase;
                    if (parsed.registrationOrders) registrationOrders = parsed.registrationOrders;
                    if (parsed.chairPaymentRecords) chairPaymentRecords = parsed.chairPaymentRecords;
                    if (parsed.emailInboxLogs) emailInboxLogs = parsed.emailInboxLogs;
                    if (parsed.chairConfig) {
                        chairConfig = Object.assign({
                            feeCompete: 1200,
                            feeMealOnly: 1000
                        }, parsed.chairConfig);
                    }
                }
                // 確保地區年會存在於 categoriesData 中
                if (!categoriesData.some(c => c.id === 'conference' || (c.name && (c.name.includes('地區年會') || c.name.includes('年會'))))) {
                    const defaultConf = DEFAULT_CATEGORIES.find(c => c.id === 'conference');
                    if (defaultConf) categoriesData.push(JSON.parse(JSON.stringify(defaultConf)));
                }
            } catch (e) {
                console.warn("讀取 LocalStorage 失敗", e);
            }
        }
